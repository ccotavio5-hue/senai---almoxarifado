CREATE DATABASE tcc;
USE tcc;

CREATE TABLE estoque (
    id INT AUTO_INCREMENT PRIMARY KEY,
    item VARCHAR(255),
    descricao VARCHAR(255),
    quantidade INT,
    imagem VARCHAR(255)
);

CREATE TABLE historico (
    id INT AUTO_INCREMENT PRIMARY KEY,
    item VARCHAR(255),
    quantidade INT,
    pessoa VARCHAR(255),
    data_hora DATETIME DEFAULT now()
);

CREATE TABLE administrador (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario VARCHAR(180) NOT NULL,
    senha VARCHAR(500)

);
